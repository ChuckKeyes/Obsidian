# Generate the private key
resource "tls_private_key" "MyLinuxBox" {
  algorithm = "RSA"
  rsa_bits  = 2048
}

# Create AWS key pair
resource "aws_key_pair" "MyLinuxBox" {
  key_name   = "MyLinuxBox-key"
  public_key = tls_private_key.MyLinuxBox.public_key_openssh
}

# Save private key locally
resource "local_file" "private_key" {
  content         = tls_private_key.MyLinuxBox.private_key_pem
  filename        = "${path.module}/MyLinuxBox-key.pem"
  file_permission = "0400"
}

# Output key name
output "key_pair_name" {
  value = aws_key_pair.MyLinuxBox.key_name
}
