resource "aws_launch_template" "web" {
  name_prefix   = "ck-web-lt-"
  image_id      = "ami-0fa3fe0fa7920f68e"   # Debian/Ubuntu or Amazon Linux in your region
  instance_type = "t3.micro"

  vpc_security_group_ids = [aws_security_group.targets_sg.id]

  user_data = base64encode(<<-EOF
    #!/bin/bash
    set -xe

    # --- Install web server (Apache/HTTPD) ---
    if command -v yum >/dev/null 2>&1; then
      yum update -y
      yum install -y httpd wget tar
      systemctl enable httpd
      systemctl start httpd
      WEBROOT="/var/www/html"
    elif command -v apt-get >/dev/null 2>&1; then
      apt-get update -y
      apt-get install -y apache2 wget tar
      systemctl enable apache2
      systemctl start apache2
      WEBROOT="/var/www/html"
    else
      WEBROOT="/var/www/html"
    fi

    # --- Simple HTML page with instance + owner + bucket ---
    cat <<HTML > $${WEBROOT}/index.html
    <html>
      <head>
        <title>ck-lab ASG instance</title>
      </head>
      <body>
        <h1>ck-lab ASG instance: $(hostname -f)</h1>
        <h2>Owner: Charles Keyes Sr.</h2>
        <p>S3 bucket: KeyesCloudSolutions</p>
      </body>
    </html>
    HTML

    # --- Install node_exporter on each web VM ---
    NODE_VERSION="1.8.2"
    cd /opt
    wget -q https://github.com/prometheus/node_exporter/releases/download/v$NODE_VERSION/node_exporter-$NODE_VERSION.linux-amd64.tar.gz
    tar xzf node_exporter-$NODE_VERSION.linux-amd64.tar.gz
    cp node_exporter-$NODE_VERSION.linux-amd64/node_exporter /usr/local/bin/
    rm -rf node_exporter-$NODE_VERSION.linux-amd64*

    # --- systemd service for node_exporter ---
    cat <<'NODEUNIT' > /etc/systemd/system/node_exporter.service
    [Unit]
    Description=Node Exporter
    Wants=network-online.target
    After=network-online.target

    [Service]
    User=root
    Group=root
    Type=simple
    ExecStart=/usr/local/bin/node_exporter

    Restart=on-failure

    [Install]
    WantedBy=multi-user.target
    NODEUNIT

    systemctl daemon-reload
    systemctl enable node_exporter
    systemctl start node_exporter
  EOF
  )
}
