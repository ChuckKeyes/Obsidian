#############################################
# IAM role + instance profile for web ASG instances
#############################################

resource "aws_iam_role" "web_role" {
  name = "ck-web-asg-role"

  assume_role_policy = <<EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Principal": {
        "Service": "ec2.amazonaws.com"
      },
      "Action": "sts:AssumeRole"
    }
  ]
}
EOF
}

resource "aws_iam_policy" "web_s3_read_images" {
  name        = "ck-web-s3-read-images"
  description = "Allow web instances to read image objects from S3"

  policy = <<EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Action": [
        "s3:GetObject"
      ],
      "Resource": "arn:aws:s3:::keyescloudsolutions/images/*"
    }
  ]
}
EOF
}

resource "aws_iam_role_policy_attachment" "web_s3_read_attach" {
  role       = aws_iam_role.web_role.name
  policy_arn = aws_iam_policy.web_s3_read_images.arn
}

resource "aws_iam_instance_profile" "web_profile" {
  name = "ck-web-asg-instance-profile"
  role = aws_iam_role.web_role.name
}
